"Tata Altroz", "Honda City", "Toyota Glanza", "Mercedes G Class" 
]; 
let bookings = JSON.parse(localStorage.getItem("vehicleBookings")) || []; 
let editingIndex = -1; 
function getAvailableVehicles(editingVehicle = null) { 
const booked = bookings 
.map((b, i) => (i !== editingIndex ? b.vehicle : null)) 
.filter(Boolean); 
return allVehicles.filter(v => !booked.includes(v) || v === editingVehicle);

opt.value = v;

function bookVehicle() {

const vehicle = document.getElementById("vehicle").value; 
const date = document.getElementById("bookingDate").value; 
const time = document.getElementById("bookingTime").value;

alert("Please fill all fields."); 
return; 
} 
if (editingIndex >= 0) { 
bookings[editingIndex] = { customerName: name, vehicle, date, time }; 
editingIndex = -1; 
} else { 
bookings.push({ customerName: name, vehicle, date, time }); 
} 
localStorage.setItem("vehicleBookings", JSON.stringify(bookings)); 
alert("Booking saved!"); 
resetForm(); 
renderCustomerBookings();

function editBooking(index) { 
const b = bookings[index]; 
const currentName = 
document.getElementById("customerName").value.trim(); 
if (b.customerName !== currentName) { 
alert("You can only edit your own bookings."); 
return; 
} 
document.getElementById("vehicle").value = b.vehicle; 
document.getElementById("bookingDate").value = b.date; 
document.getElementById("bookingTime").value = b.time; 
editingIndex = index; 
populateDropdown(b.vehicle); 
} 
function deleteBooking(index) { 
const currentName = 
document.getElementById("customerName").value.trim(); 
if (bookings[index].customerName !== currentName) { 
alert("You can only delete your own bookings."); 
return; 
} 
if (confirm("Are you sure you want to delete your booking?")) { 
bookings.splice(index, 1); 
localStorage.setItem("vehicleBookings", JSON.stringify(bookings));
